# requirements.lock.txt - pinned, reproducible Python dependency set.
#
# TARGET RUNTIME: CPython 3.10 or 3.11. All pins below were chosen for that
# target. The Mac .venv this pilot runs in is 3.10.
#
# NUMPY: the `numpy>=1.26,<2.0` ceiling in requirements.txt WINS; this lock
# conforms to it rather than silently overriding the app's stated constraint.
#   * numpy 2.0 broke the C ABI: any extension compiled against numpy 1.x
#     aborts at import under numpy 2.x. Pinning the whole numpy-1.x generation
#     removes that runtime hazard outright.
#   * easyocr 1.7.2 (the stack's laggard) and several compiled transitives are
#     only reliably exercised on numpy 1.x.
#   * the packages pinned below form a numpy-1.x-consistent set: torchvision
#     0.19.1 requires torch==2.4.1; ultralytics core requires numpy>=1.23.0
#     (and pins numpy<2 on darwin); opencv 4.10 requires numpy>=1.23.5.
#     numpy 1.26.4 satisfies all of them.
#
# USAGE (reproducible install):
#   pip install -r requirements.lock.txt
#
# TRANSITIVES: this file pins the DIRECT dependencies exactly. To produce a
#   fully transitive-pinned lock, run on a clean target interpreter:
#   pip install -r requirements.lock.txt && pip freeze > requirements.lock.txt

# --- CV pipeline (pipeline/) ---
opencv-python-headless==4.10.0.84   # headless (no GUI libs)
numpy==1.26.4                        # constraint winner: satisfies requirements.txt <2.0

                                     # (!=8.3.*,>=5.3.0), ultralytics 8.3.43 (>=7.1.2) and easyocr 1.7.2 all accept it.
pillow-heif==1.7.0                   # HEIC/AVIF decode for pipeline/normalize_image.py; wheels bundle libheif

torchvision==0.19.1
